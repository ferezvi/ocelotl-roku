import "pkg:/source/utils/misc.bs"

const SCREEN_WIDTH = 1920

' Fixed preview box (object-fit: contain).
const TRICKPLAY_BOX_WIDTH = 320
const TRICKPLAY_BOX_HEIGHT = 180

' Maximum decoded mosaic dimensions.
'
' This controls texture memory only.
' It does NOT change the visual size of the preview.
const TRICKPLAY_MAX_MOSAIC_WIDTH = 1600
const TRICKPLAY_MAX_MOSAIC_HEIGHT = 1200


sub init()
    m.trickplayImageA = m.top.findNode("trickplayImageA")
    m.trickplayImageB = m.top.findNode("trickplayImageB")
    m.trickplayBackground = m.top.findNode("trickplayBackground")

    m.trickplayBackground.width = TRICKPLAY_BOX_WIDTH
    m.trickplayBackground.height = TRICKPLAY_BOX_HEIGHT

    ' A starts as the active buffer.
    m.activeImage = m.trickplayImageA
    m.loadingImage = m.trickplayImageB

    ' True when the loading buffer is A.
    ' False when the loading buffer is B.
    m.loadingIsA = false

    m.activeTileIndex = -1
    m.loadingTileIndex = -1

    m.activeTileUri = ""
    m.loadingTileUri = ""

    m.requestedTileIndex = -1
    m.requestedTileUri = ""

    m.top.observeField("position", "onPositionChanged")
    m.top.observeField("trickplayData", "onTrickplayDataChanged")

    m.trickplayImageA.observeField("loadStatus", "onImageALoadStatus")
    m.trickplayImageB.observeField("loadStatus", "onImageBLoadStatus")
end sub


sub onPositionChanged()
    if not isValid(m.top.trickplayData) then return
    if not isValid(m.top.position) then return

    displayTrickplayImage(m.top.position)
end sub


sub onTrickplayDataChanged()
    if not isValid(m.top.trickplayData)
        hide()
        return
    end if

    m.activeTileIndex = -1
    m.loadingTileIndex = -1

    m.activeTileUri = ""
    m.loadingTileUri = ""

    m.requestedTileIndex = -1
    m.requestedTileUri = ""

    m.activeImage = m.trickplayImageA
    m.loadingImage = m.trickplayImageB
    m.loadingIsA = false

    m.trickplayImageA.visible = false
    m.trickplayImageB.visible = false

    m.trickplayImageA.uri = ""
    m.trickplayImageB.uri = ""

    if isValid(m.top.position)
        displayTrickplayImage(m.top.position)
    end if
end sub


sub displayTrickplayImage(position as float)
    data = m.top.trickplayData

    if not isValid(data) then return

    interval = data.Interval / 1000.0

    if interval <= 0 then return

    iconIndex = fix(position / interval)

    tilesPerImage = data.TileHeight * data.TileWidth

    if tilesPerImage <= 0 then return

    tileIndex = fix(iconIndex / tilesPerImage)
    tileIconIndex = iconIndex - (tileIndex * tilesPerImage)

    iconRow = fix(tileIconIndex / data.TileWidth)
    iconColumn = tileIconIndex mod data.TileWidth

    ' ------------------------------------------------------------
    ' DISPLAY SCALE
    '
    ' Controls only the final on-screen size.
    ' ------------------------------------------------------------

    displayScale = TRICKPLAY_BOX_HEIGHT / data.Height
    displayScaleAlt = TRICKPLAY_BOX_WIDTH / data.Width

    if displayScaleAlt < displayScale
        displayScale = displayScaleAlt
    end if

    tileWidth = data.Width * displayScale
    tileHeight = data.Height * displayScale

    mosaicWidth = data.Width * data.TileWidth * displayScale
    mosaicHeight = data.Height * data.TileHeight * displayScale

    ' ------------------------------------------------------------
    ' LOAD SCALE
    '
    ' Controls only decoded texture size.
    ' ------------------------------------------------------------

    loadMosaicWidth = mosaicWidth
    loadMosaicHeight = mosaicHeight

    loadCapScale = TRICKPLAY_MAX_MOSAIC_WIDTH / mosaicWidth
    loadCapScaleAlt = TRICKPLAY_MAX_MOSAIC_HEIGHT / mosaicHeight

    if loadCapScaleAlt < loadCapScale
        loadCapScale = loadCapScaleAlt
    end if

    if loadCapScale < 1.0
        loadMosaicWidth = mosaicWidth * loadCapScale
        loadMosaicHeight = mosaicHeight * loadCapScale
    else
        loadCapScale = 1.0
    end if

    tileUri = "tmp:/" + m.top.videoId + "-" + tileIndex.ToStr() + ".jpg"

    ' ------------------------------------------------------------
    ' PREVIEW POSITION
    ' ------------------------------------------------------------

    previewX = m.top.previewX

    sideMargin = (SCREEN_WIDTH - m.top.progressBarWidth) / 2
    boxLeft = sideMargin + previewX - (TRICKPLAY_BOX_WIDTH / 2)

    ' Center selected tile inside fixed preview box.
    barX = (TRICKPLAY_BOX_WIDTH - tileWidth) / 2
    barY = (TRICKPLAY_BOX_HEIGHT - tileHeight) / 2

    imageTranslation = [
        boxLeft + barX - (iconColumn * tileWidth),
        barY - (iconRow * tileHeight)
    ]

    ' ------------------------------------------------------------
    ' IMPORTANT:
    '
    ' clippingRect is expressed in the DECODED TEXTURE coordinate
    ' system, not in the final display coordinate system.
    '
    ' Therefore, when the texture is decoded smaller, the clipping
    ' rectangle must be scaled by loadCapScale as well.
    ' ------------------------------------------------------------

    loadTileWidth = tileWidth * loadCapScale
    loadTileHeight = tileHeight * loadCapScale

    imageClippingRect = [
        iconColumn * loadTileWidth,
        iconRow * loadTileHeight,
        loadTileWidth,
        loadTileHeight
    ]

    ' ------------------------------------------------------------
    ' Configure BOTH buffers geometrically.
    '
    ' This is intentional and preserves the behavior of the
    ' double-buffer version that was visually smooth.
    ' ------------------------------------------------------------

    configureImage(
        m.trickplayImageA,
        mosaicWidth,
        mosaicHeight,
        imageTranslation,
        imageClippingRect,
        loadMosaicWidth,
        loadMosaicHeight
    )

    configureImage(
        m.trickplayImageB,
        mosaicWidth,
        mosaicHeight,
        imageTranslation,
        imageClippingRect,
        loadMosaicWidth,
        loadMosaicHeight
    )

    m.trickplayBackground.translation = [boxLeft, 0]
    m.trickplayBackground.visible = true

    ' ------------------------------------------------------------
    ' REQUESTED TILE IS ALREADY ACTIVE.
    ' ------------------------------------------------------------

    if tileIndex = m.activeTileIndex
        m.requestedTileIndex = tileIndex
        m.requestedTileUri = tileUri
        return
    end if

    ' ------------------------------------------------------------
    ' REQUESTED TILE IS ALREADY LOADING.
    ' ------------------------------------------------------------

    if tileIndex = m.loadingTileIndex and tileUri = m.loadingTileUri
        m.requestedTileIndex = tileIndex
        m.requestedTileUri = tileUri

        if m.loadingImage.loadStatus = "ready"
            onLoadingImageReady()
        end if

        return
    end if

    ' ------------------------------------------------------------
    ' REQUESTED URI ALREADY LOADED IN BUFFER A.
    ' ------------------------------------------------------------

    if m.trickplayImageA.uri = tileUri and m.trickplayImageA.loadStatus = "ready"

        if m.loadingIsA
            m.loadingTileIndex = tileIndex
            m.loadingTileUri = tileUri

            m.requestedTileIndex = tileIndex
            m.requestedTileUri = tileUri

            onLoadingImageReady()
        else
            m.activeTileIndex = tileIndex
            m.activeTileUri = tileUri

            m.requestedTileIndex = tileIndex
            m.requestedTileUri = tileUri
        end if

        return
    end if

    ' ------------------------------------------------------------
    ' REQUESTED URI ALREADY LOADED IN BUFFER B.
    ' ------------------------------------------------------------

    if m.trickplayImageB.uri = tileUri and m.trickplayImageB.loadStatus = "ready"

        if not m.loadingIsA
            m.loadingTileIndex = tileIndex
            m.loadingTileUri = tileUri

            m.requestedTileIndex = tileIndex
            m.requestedTileUri = tileUri

            onLoadingImageReady()
        else
            m.activeTileIndex = tileIndex
            m.activeTileUri = tileUri

            m.requestedTileIndex = tileIndex
            m.requestedTileUri = tileUri
        end if

        return
    end if

    ' ------------------------------------------------------------
    ' NEW MOSAIC.
    '
    ' NEVER change the active URI.
    ' Only the inactive buffer receives the new URI.
    ' ------------------------------------------------------------

    m.requestedTileIndex = tileIndex
    m.requestedTileUri = tileUri

    m.loadingTileIndex = tileIndex
    m.loadingTileUri = tileUri

    m.loadingImage.visible = false

    ' loadWidth/loadHeight MUST be assigned before URI.
    m.loadingImage.loadWidth = loadMosaicWidth
    m.loadingImage.loadHeight = loadMosaicHeight

    if m.loadingImage.uri <> tileUri
        m.loadingImage.uri = tileUri
    else if m.loadingImage.loadStatus = "ready"
        onLoadingImageReady()
    end if
end sub


sub configureImage(image as object, mosaicWidth as float, mosaicHeight as float, translation as object, clippingRect as object, loadWidth as float, loadHeight as float)
    if not isValid(image) then return

    ' Decode dimensions.
    image.loadWidth = loadWidth
    image.loadHeight = loadHeight

    ' Display dimensions.
    image.width = mosaicWidth
    image.height = mosaicHeight

    image.translation = translation
    image.clippingRect = clippingRect
end sub


sub onImageALoadStatus(event as object)
    if not isValid(event) then return

    if event.getData() <> "ready" then return

    if not m.loadingIsA then return

    onLoadingImageReady()
end sub


sub onImageBLoadStatus(event as object)
    if not isValid(event) then return

    if event.getData() <> "ready" then return

    if m.loadingIsA then return

    onLoadingImageReady()
end sub


sub onLoadingImageReady()
    if not isValid(m.loadingImage) then return

    if m.loadingTileIndex < 0 then return

    ' User moved again while the image was loading.
    if m.loadingTileIndex <> m.requestedTileIndex
        return
    end if

    if m.loadingTileUri <> m.requestedTileUri
        return
    end if

    if m.loadingImage.loadStatus <> "ready"
        return
    end if

    swapImages()
end sub


sub swapImages()
    if m.loadingTileIndex < 0 then return

    ' Keep the original swap behavior because this is the version
    ' that produced the least noticeable transition.
    m.loadingImage.visible = true
    m.activeImage.visible = false

    oldActiveImage = m.activeImage

    m.activeImage = m.loadingImage
    m.loadingImage = oldActiveImage

    m.loadingIsA = not m.loadingIsA

    m.activeTileIndex = m.loadingTileIndex
    m.activeTileUri = m.loadingTileUri

    m.loadingTileIndex = -1
    m.loadingTileUri = ""

    m.requestedTileIndex = m.activeTileIndex
    m.requestedTileUri = m.activeTileUri
end sub


sub hide()
    m.trickplayImageA.visible = false
    m.trickplayImageB.visible = false
    m.trickplayBackground.visible = false
end sub
